\section{INTRODUCCION}
% ============================================================

\begin{frame}{Metodologia: Multiple Shooting Directo}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{tbox}{Principio de Multiple Shooting}
\small
\begin{itemize}
\item Se discretiza el horizonte temporal en $N$ intervalos $[t_k, t_{k+1}]$ de duracion $h_k$.
\item Estados $\bx_k$ y controles $\bu_k$ son variables de decision independientes en el NLP.
\item \textbf{Continuidad dinamica:}
\begin{equation*}
\bx_{k+1} - \bm{\Psi}_k(\bx_k, \bu_k, h_k) = \mathbf{0}
\end{equation*}
donde $\bm{\Psi}_k$ integra el sistema mediante \textbf{RK4} en subpasos de $h_k/2$.
\end{itemize}
\end{tbox}
\end{column}

\begin{column}{0.48\textwidth}
\begin{tbox}{Ventajas Computacionales}
\small
\begin{itemize}
\item \textbf{Estabilidad numerica:} evita la divergencia por integracion en horizontes largos (problema tipico de \textit{single shooting}).
\item \textbf{Esparsidad estructural:} Jacobiano y Hessiano altamente esparsos, optimos para solucionadores NLP como \textbf{IPOPT}.
\item \textbf{Restricciones intermedias:} acoplamiento directo de eventos hibridos (contactos, cruces y geometrias complejas).
\end{itemize}
\end{tbox}
\end{column}
\end{columns}
\src{CasADi + IPOPT con estrategia de barrera adaptativa}
\end{frame}

\section{CUADRUPEDO: SALTO}
% ============================================================

\begin{frame}{P1: Modelo Dinamico del Cuadrupedo (SRBM)}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\begin{tbox}{Estado y Entradas}
\footnotesize
\textbf{Estado} $\bx \in \RR^{12}$ y \textbf{Control} $\bu \in \RR^{12}$:
\begin{align*}
\bx &= \begin{bmatrix}\bp\tr & \bm{\eta}\tr & \bv\tr & {}^b\bomega\tr\end{bmatrix}\tr\\
\bu &= \begin{bmatrix}\bef_{FR}\tr & \bef_{FL}\tr & \bef_{RR}\tr & \bef_{RL}\tr\end{bmatrix}\tr
\end{align*}
\begin{itemize}
\item $\bp=(x,y,z)$: posicion del CdM en $\{W\}$.
\item $\bm{\eta}=(\phi,\theta,\psi)$: orientacion roll--pitch--yaw.
\item $\bef_i$: fuerza de contacto del suelo sobre el pie $i$.
\end{itemize}
\end{tbox}
\end{column}

\begin{column}{0.46\textwidth}
\begin{tbox}{Ecuaciones de Movimiento}
\footnotesize
\begin{align*}
\dot{\bp} &= \bv \\
\dot{\bv} &= \mathbf{g} + \frac{1}{m}\sum_{i=1}^4 \bef_i \\
\dot{\bm{\eta}} &= E(\phi,\theta)\,{}^b\bomega \\
I\,{}^b\dot{\bomega} &= {}^w\bR_b\tr \sum_{i=1}^4 (\bp_i - \bp)\times \bef_i - {}^b\bomega \times (I\,{}^b\bomega)
\end{align*}
Patas modeladas sin masa (SRBM); parametros leidos directamente del URDF del Go2.
\end{tbox}
\end{column}
\end{columns}
\src{Cinematica rotacional con matriz $E(\phi,\theta)$ y orientacion $\bR = \bR_z(\psi)\bR_y(\theta)\bR_x(\phi)$}
\end{frame}

\begin{frame}{P1: Secuencia de Fases y Contactos}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{tbox}{Estructura de 5 Fases ($N=88$ nodos)}
\scriptsize
\resizebox{\linewidth}{!}{%
\begin{tabular}{@{}clccc@{}}
\toprule
$j$ & Descripcion & Apoyos $\mathcal{S}_j$ & $N_j$ & $T_j$ (s) \\
\midrule
1 & Impulso 4 patas & FR, FL, RR, RL & 22 & $[0.25,\,0.80]$ \\
2 & Impulso traseras & RR, RL & 10 & $[0.05,\,0.15]$ \\
3 & Vuelo libre & $\emptyset$ & 24 & $[0.15,\,1.70]$ \\
4 & Aterrizaje delanteras & FR, FL & 10 & $[0.05,\,0.15]$ \\
5 & Apoyo 4 patas & FR, FL, RR, RL & 22 & $[0.25,\,0.80]$ \\
\bottomrule
\end{tabular}}
\vskip4pt
Las duraciones $T_j$ son variables de decision acotadas: $h_k = T_{j(k)}/N_{j(k)}$.
\end{tbox}
\end{column}

\begin{column}{0.48\textwidth}
\begin{tbox}{Restricciones Fisicas de Contacto}
\small
\begin{itemize}
\item \textbf{Friccion y fuerza normal:}
\begin{gather*}
0 \le f_{z,i} \le 220\,\text{N}\\
|f_{x,i}| \le \mu f_{z,i},\quad |f_{y,i}| \le \mu f_{z,i}
\end{gather*}
con coeficiente de friccion $\mu = 0.70$.
\item \textbf{Pata en vuelo:} $\bef_i = \mathbf{0}$ si $i \notin \mathcal{S}_{j(k)}$.
\item \textbf{Alcance de pata:} $0.20 \le \|\bd_i\| \le 0.39$\,m, donde $\bd_i$ es el vector cadera--pie en $\{B\}$.
\end{itemize}
\end{tbox}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{P1: Obstaculo Suave y Optimizacion NLP}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{tbox}{Obstaculo Diferenciable}
\small
Caja de $H=0.25$\,m en $x \in [0.42, 0.67]$\,m aproximada mediante producto de sigmoides:
\begin{equation*}
\begin{aligned}
h_{\mathrm{obs}}(x) = (H+\delta)\,&\sigma\big(k(x-x_1+\delta)\big)\\
\cdot\,&\sigma\big(k(x_2+\delta-x)\big)
\end{aligned}
\end{equation*}
\begin{itemize}
\item Margen de inflado: $\delta = 3$\,cm, $k=80$.
\item Se exige $z_c \ge h_{\mathrm{obs}}(x_c)$ para esquinas del tronco, rodillas y pies durante todo el salto.
\end{itemize}
\end{tbox}
\end{column}

\begin{column}{0.48\textwidth}
\begin{tbox}{Problema NLP y Costo}
\small
\begin{equation*}
J = \sum_{k=0}^{N-1} h_k \ell_k + 0.02 \sum_{k=1}^{N-1} \Big\|\frac{\Delta \bu_k}{mg}\Big\|^2
\end{equation*}
\begin{equation*}
\ell_k = \Big\|\frac{\bu_k - \bu_k^{\mathrm{est}}}{mg}\Big\|^2 + 0.05\|{}^b\bomega_k\|^2 + 2(\phi_k^2 + \psi_k^2)
\end{equation*}
\begin{itemize}
\item Penaliza desviacion del apoyo estatico $\bu_k^{\mathrm{est}}$.
\item Regulariza variacion temporal de fuerzas.
\end{itemize}
\end{tbox}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{P1: Cinematica Inversa y Torques Articulares}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{tbox}{Cinematica Inversa Analitica}
\small
Para cada pata $i$ ($q_0$: abduccion, $q_1$: cadera, $q_2$: rodilla):
\begin{itemize}
\item \textbf{En apoyo:} posicion del pie fija en el suelo:
\begin{equation*}
\bm{q}_i = \mathrm{CI}_i\big({}^w\bR_b\tr(\bp_i^c - \bp)\big)
\end{equation*}
\item \textbf{En vuelo:} interpolacion quintica articular suave entre despegue, pose recogida y aterrizaje.
\end{itemize}
\end{tbox}
\end{column}

\begin{column}{0.48\textwidth}
\begin{tbox}{Mapeo a Torques de Motores}
\small
Los torques requeridos en los motores para producir las fuerzas optimas de contacto se obtienen via el \textbf{Jacobiano}:
\begin{equation*}
\btau_i = -J_i(\bm{q}_i)\tr\,{}^w\bR_b\tr\,\bef_i
\end{equation*}
\begin{itemize}
\item Permite validar que los actuadores no superen limites termicos ni de torque maximo.
\end{itemize}
\end{tbox}
\end{column}
\end{columns}
\end{frame}

\section{QUADCÓPTERO: SNAP ROLL}
% ============================================================

\begin{frame}{P2: Modelo Dinamico del Quadcoptero}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\begin{tbox}{Estados y Actuacion}
\footnotesize
\textbf{Estado} $\bx \in \RR^{12}$ y \textbf{Control} $\bu \in \RR^4$:
\begin{align*}
\bx &= \begin{bmatrix}\bp\tr & \bm{\eta}\tr & \bv\tr & \bomega\tr\end{bmatrix}\tr\\
\bu &= \begin{bmatrix}F & \tau_x & \tau_y & \tau_z\end{bmatrix}\tr
\end{align*}
\begin{itemize}
\item $F$: empuje total colectivo a lo largo del eje $z$ del cuerpo.
\item $\btau = (\tau_x,\tau_y,\tau_z)$: momentos de control.
\item Parametros: $m=1$\,kg, $I=\mathrm{diag}(0.02, 0.02, 0.04)\,\text{kg}\cdot\text{m}^2$.
\end{itemize}
\end{tbox}
\end{column}

\begin{column}{0.46\textwidth}
\begin{tbox}{Dinamica SE(3)}
\footnotesize
\begin{align*}
\dot{\bp} &= \bv \\
m\dot{\bv} &= F\,\bR(\bm{\eta})\mathbf{e}_3 - mg\,\mathbf{e}_3 \\
\dot{\bm{\eta}} &= E(\phi,\theta)\,\bomega \\
I\dot{\bomega} &= \btau - \bomega \times (I\bomega)
\end{align*}
\begin{itemize}
\item Orientacion por angulos Euler ZYX.
\item Se acota $|\theta| \le \theta_{\max} = 0.55$\,rad ($\approx 31.5^\circ$) para evitar la singularidad del gimbal lock.
\end{itemize}
\end{tbox}
\end{column}
\end{columns}
\src{Condiciones: $\bp_0=(0,0,1)$\,m a $\bp_f=(4,2,1.3)$\,m en $T=6$\,s con $\psi_f=90^\circ$}
\end{frame}

\begin{frame}{P2: Geometria de la Pared y Maniobra Invertida}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\begin{tbox}{Geometria de la Ventana}
\small
Pared en $x_w=2$\,m (espesor $e=0.06$\,m) con orificio centrado en $\bm{c}_h = (2, 1, 1)$\,m:
\begin{equation*}
|y - y_h| \le a,\quad |z - z_h| \le b
\end{equation*}
con semiancho $a=0.32$\,m y semialtura $b=0.25$\,m.
\vskip4pt
Brazos del quad: $\ell=0.17$\,m, radio rotores $r_r=0.03$\,m.
\end{tbox}
\end{column}

\begin{column}{0.46\textwidth}
\begin{tbox}{Ventaja del Snap Roll}
\small
\begin{itemize}
\item Al cruzar boca abajo ($\phi = 180^\circ = \pi$):
\begin{equation*}
\bm{b}_2 \cdot \mathbf{e}_3 \approx 0
\end{equation*}
\item El brazo transversal se alinea horizontalmente con la apertura ancha ($a=0.32$\,m).
\item Permite atravesar un hueco de solo $0.50$\,m de altura total sin colision.
\end{itemize}
\end{tbox}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{P2: Formulacion del NLP (Multiple Shooting)}
\begin{tbox}{Malla Discreta y Restricciones de Cruce ($N=70$ intervalos, $T=6$\,s)}
\small
\begin{itemize}
\item \textbf{Nodo de cruce:} $k_c = N/2 = 35$ ($t_c = 3.0$\,s).
\item \textbf{Cruce estricto invertido:} $\phi_{k_c} = \pi$, $x_{k_c} = x_w = 2.0$\,m, y velocidad minima de paso $v_{x,k_c} \ge 2.4$\,m/s.
\item \textbf{Paso asegurado entre caras:} $x_{k_c-1} \le x_w - e/2$ y $x_{k_c+1} \ge x_w + e/2$.
\item \textbf{Holguras de las 4 puntas de rotor} con margen de seguridad $\delta=0.03$\,m:
\begin{equation*}
\big|(y - y_h) \pm \ell\,\bm{b}_i \cdot \mathbf{e}_2\big| \le a - r_r - \delta, \qquad
\big|(z - z_h) \pm \ell\,\bm{b}_i \cdot \mathbf{e}_3\big| \le b - r_r - \delta
\end{equation*}
\end{itemize}
\end{tbox}
\vskip6pt
\begin{pbox}
\footnotesize
\textbf{Funcion de Costo:} Penaliza desviaciones de trayectoria de referencia (curvas quinticas / Hermite), minimiza esfuerzo de empuje $(F-mg)^2$, momento $\|\btau\|^2$ y variaciones suaves de control $\|\Delta \bu\|^2$.
\end{pbox}
\end{frame}

\begin{frame}{Conclusiones}
\begin{tbox}{Aportes Principales}
\small
\begin{itemize}
\item \textbf{Eficacia de Multiple Shooting:} Probo ser altamente superior frente a esquemas de tiro simple, resolviendo problemas no lineales severos y no convexos con convergencia garantizada en IPOPT.
\item \textbf{Manejo de Dinamicas Hibridas:} Permitio optimizar conjuntamente la duracion de cada fase y la ubicacion de pisada en el robot cuadrúpedo, superando un obstaculo inerte de $0.25$\,m sin requerir contacto sobre el.
\item \textbf{Exploracion de Maniobras Extremas:} La orientacion acrobática de $180^\circ$ en el quadcoptero explota la geometria del vehiculo para franquear un orificio geometricamente limitante.
\item \textbf{Fidelidad Dinamica:} Las simulaciones de integracion continua (DOP853) confirman la viabilidad fisica de las trayectorias discretizadas obtenidas.
\end{itemize}
\end{tbox}
